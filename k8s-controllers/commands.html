🔧 Kubernetes Commands

📦 Deployment Controller Commands
✅ Correct (from YAML): webapp-deployment
# Create deployment
kubectl apply -f 01-deployment-controller.yaml

# Get deployment status
kubectl get deployments
kubectl get deployment webapp-deployment -o wide

# Describe deployment
kubectl describe deployment webapp-deployment

# Scale deployment
kubectl scale deployment webapp-deployment --replicas=5

# Rolling update
kubectl set image deployment/webapp-deployment webapp=ivykiera/maven-web-app:v2

# Check rollout status
kubectl rollout status deployment/webapp-deployment

# View rollout history
kubectl rollout history deployment/webapp-deployment

# Rollback to previous version
kubectl rollout undo deployment/webapp-deployment

# Get pods
kubectl get pods -l app=webapp

🗄️ StatefulSet Controller Commands
✅ Name: postgres-statefulset

kubectl apply -f 04-statefulset-controller.yaml

# Get StatefulSet status
kubectl get statefulsets
kubectl get sts postgres-statefulset -o wide

# Describe StatefulSet
kubectl describe statefulset postgres-statefulset

# Get pods (ordered names)
kubectl get pods -l app=postgres

kubectl scale statefulset postgres-statefulset --replicas=5

# Connect to specific pod
kubectl exec -it postgres-statefulset-0 -- psql -U postgres -d mydb

⏰ CronJob Controller Commands
✅ Name: backup-cronjob
# Create CronJob
kubectl apply -f 06-cronjob-controller.yaml

# Get CronJob status
kubectl get cronjobs
kubectl get cj backup-cronjob -o wide

# Describe CronJob
kubectl describe cronjob backup-cronjob

# Get jobs created by CronJob
kubectl get jobs -l app=backup

# Manually trigger CronJob
kubectl create job manual-backup --from=cronjob/backup-cronjob

# Suspend CronJob
kubectl patch cronjob backup-cronjob -p '{"spec":{"suspend":true}}'
📈 HPA Controller Commands
✅ webapp-hpa (targets webapp-deployment)

kubectl apply -f 07-hpa-controller.yaml
